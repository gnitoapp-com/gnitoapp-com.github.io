{{- /* Product landing page layout. All shared chrome (header, quick info, integrity block, install guide, screenshots) is rendered from data/products/.yaml plus the generated data/apk_checksums.json. The page body holds only the product's description. Required front matter: layout: product, product: . */ -}} {{- define "main" -}} {{- $product := .Params.product -}} {{- if not $product -}} {{ errorf "%s: front matter needs `product:` (see data/products/)" .File.Path }} {{- end -}} {{- $data := index hugo.Data.products $product -}} {{- if not $data -}} {{ errorf "%s: data/products/%s.yaml is missing" .File.Path $product }} {{- end -}} {{- if not $data.platforms -}} {{ errorf "%s: data/products/%s.yaml needs a `platforms:` list (e.g. [android, ios])" .File.Path $product }} {{- end -}} {{- $android := in $data.platforms "android" -}} {{- $apk := false -}} {{- if $android -}} {{- $apk = index hugo.Data.apk_checksums $product -}} {{- if not $apk -}} {{ errorf "%s: %s is marked Android but has no entry in data/apk_checksums.json - run scripts/apk-checksums.sh" .File.Path $product }} {{- end -}} {{- end -}} {{- /* Desktop installers: resolve each entry in `downloads:` to the file name and href up front, so the header buttons and the downloads section cannot disagree. Names follow the app's packaging script: --[-]. */ -}} {{- $downloads := slice -}} {{- if $data.downloads -}} {{- if not $data.version -}} {{- errorf "%s: data/products/%s.yaml has `downloads:` but no `version:`" .File.Path $product -}} {{- end -}} {{- range $data.downloads -}} {{- if not (in $data.platforms .platform) -}} {{- errorf "%s: data/products/%s.yaml lists a download for %q, which is not in `platforms:`" $.File.Path $product .platform -}} {{- end -}} {{- $arch := "" -}} {{- with .arch }}{{ $arch = printf "-%s" . }}{{ end -}} {{- $file := printf "%s-%s-%s%s.%s" $product $data.version .platform $arch .ext -}} {{- $downloads = $downloads | append (merge . (dict "file" $file "href" (printf "/%s/%s" $product $file))) -}} {{- end -}} {{- end -}} {{- $ctx := dict "page" . "product" $product "data" $data "apk" $apk "android" $android "downloads" $downloads -}}
{{ partial "product/header.html" $ctx }} {{ partial "product/downloads.html" $ctx }} {{- /* APK-specific blocks only exist for Android products. */ -}} {{ if $android }}{{ partial "apk-verify.html" $ctx }}{{ end }} {{ partial "product/quickinfo.html" $ctx }} {{ if $android }}{{ partial "product/install.html" $ctx }}{{ end }} {{ partial "product/screenshots.html" $ctx }}
{{ .Content }}
{{- end -}}
